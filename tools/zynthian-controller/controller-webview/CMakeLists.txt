cmake_minimum_required(VERSION 3.22)
project(zynthian_controller VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network Widgets WebEngineWidgets WebChannel)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
add_library(controller-core src/model.cpp src/input.cpp)
target_include_directories(controller-core PUBLIC src)
target_link_libraries(controller-core PUBLIC Qt6::Core Qt6::Gui Qt6::Network PkgConfig::SDL2)
target_compile_options(controller-core PRIVATE -Wall -Wextra -Wpedantic)
add_library(controller-host src/host.cpp)
target_link_libraries(controller-host PUBLIC controller-core Qt6::Widgets Qt6::WebEngineWidgets Qt6::WebChannel)
target_compile_options(controller-host PRIVATE -Wall -Wextra -Wpedantic)
add_executable(zynthian-controller MACOSX_BUNDLE src/main.cpp)
target_link_libraries(zynthian-controller PRIVATE controller-host)
set_target_properties(zynthian-controller PROPERTIES
  MACOSX_BUNDLE_BUNDLE_NAME "Zynthian Controller"
  MACOSX_BUNDLE_GUI_IDENTIFIER "org.crest.zynthian-controller"
  MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
  MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(controller-tests tests/core_tests.cpp)
  target_link_libraries(controller-tests PRIVATE controller-host Qt6::Test)
  add_test(NAME controller-core COMMAND controller-tests)
  # Virtual-controller regression tests must not consume the user's live pad.
  set_tests_properties(controller-core PROPERTIES ENVIRONMENT "SDL_JOYSTICK_HIDAPI=0;SDL_JOYSTICK_HIDAPI_ONLY=1")
  add_executable(controller-live-tests tests/live_tests.cpp resources.qrc)
  target_link_libraries(controller-live-tests PRIVATE controller-core Qt6::Test Qt6::Widgets Qt6::WebEngineWidgets Qt6::WebChannel)
  # Deliberately not a default CTest: requires a disposable, running Zynthian backend.
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME zynthian-bridge COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_*.py)
endif()
install(TARGETS zynthian-controller RUNTIME DESTINATION bin BUNDLE DESTINATION .)
